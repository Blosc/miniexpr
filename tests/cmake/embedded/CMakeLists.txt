cmake_minimum_required(VERSION 3.16.3)
project(miniexpr_embedded_consumer LANGUAGES C)

if(NOT MINIEXPR_SOURCE)
  message(FATAL_ERROR "Set MINIEXPR_SOURCE to the miniexpr checkout")
endif()
add_subdirectory("${MINIEXPR_SOURCE}" miniexpr)
if(MINIEXPR_BUILD_TESTS OR MINIEXPR_BUILD_EXAMPLES OR MINIEXPR_BUILD_BENCH)
  message(FATAL_ERROR "Embedded builds must default tests, examples and benchmarks OFF")
endif()
add_executable(grid_consumer "${MINIEXPR_SOURCE}/tests/test_grid_api.c")
add_executable(pairwise_consumer "${MINIEXPR_SOURCE}/tests/test_pairwise_sum_core.c")
if(MINIEXPR_BUILD_ARTIFACT)
  target_link_libraries(grid_consumer PRIVATE miniexpr::artifact)
  target_link_libraries(pairwise_consumer PRIVATE miniexpr::artifact)
else()
  target_link_libraries(grid_consumer PRIVATE miniexpr::core)
  target_link_libraries(pairwise_consumer PRIVATE miniexpr::core)
endif()
enable_testing()
add_test(NAME embedded_grid_consumer COMMAND grid_consumer)
add_test(NAME embedded_pairwise_consumer COMMAND pairwise_consumer)
